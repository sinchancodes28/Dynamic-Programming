class Solution {
    public int minInsertions(String s) {
        // to make a string palindrome : here we need to find the mininum number of the operation on the string 
        // the number of insertion or deletion are same
        // reverse(s) = t;
        // int lps = lcs(s, reverse t)
        // return s.length - lcs(s, reverse t)
       String t =  reverse(s);
        int lps = lcs(s ,t );
        return s.length() - lps;
    }
    String reverse(String s){
        char[] arr = s.toCharArray();
        int i = 0; int j = arr.length - 1;
        while( i < j){
            char temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
            i++;
            j--;
        }
        String t = new String(arr);
        return t;
    }
    int lcs(String s , String t){
        int n = s.length();
        int m = t.length();
        int[][] dp = new int[n+1][m+1];
        for( int i = 1 ; i <= n ; i++){
            for( int j = 1; j <= m ; j++){
                if(s.charAt(i-1) == t.charAt(j-1)){
                    dp[i][j] = 1 + dp[i-1][j-1];
                }
                else{
                    dp[i][j] = Math.max(dp[i-1][j] , dp[i][j-1]);
                }
            } 
        }
        return dp[n][m];
    }

}
